cmake_minimum_required(VERSION 3.20)
project(XRayTHMTools VERSION 0.4.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
 set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
 add_compile_options(/W4 /EHsc /utf-8)
endif()
add_executable(thm_decode decode_cli.cpp thm_decoder.cpp)
add_executable(test_decoder test_decoder.cpp thm_decoder.cpp)
enable_testing()
add_test(NAME decoder COMMAND test_decoder)
add_executable(test_texture editor/test_texture.cpp editor/texture_model.cpp thm_document.cpp thm_decoder.cpp)
target_include_directories(test_texture PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
add_test(NAME texture COMMAND test_texture)
if(WIN32)
 add_executable(XRayTHMEditor WIN32 editor/editor_main.cpp editor/safe_file.cpp editor/texture_model.cpp thm_document.cpp thm_decoder.cpp)
 target_include_directories(XRayTHMEditor PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
 target_compile_definitions(XRayTHMEditor PRIVATE UNICODE _UNICODE)
 target_link_libraries(XRayTHMEditor PRIVATE user32 gdi32 ole32 shell32 comdlg32 advapi32 uuid)
 add_executable(test_safe_file editor/test_safe_file.cpp editor/safe_file.cpp)
 add_test(NAME safe_file COMMAND test_safe_file)
 add_library(XRayTHMLister SHARED lister/thm_lister.cpp thm_document.cpp thm_decoder.cpp editor/texture_model.cpp)
 target_include_directories(XRayTHMLister PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
 target_compile_definitions(XRayTHMLister PRIVATE UNICODE _UNICODE)
 target_link_libraries(XRayTHMLister PRIVATE user32 gdi32 shell32 comctl32)
 set_target_properties(XRayTHMLister PROPERTIES PREFIX "")
 if(CMAKE_SIZEOF_VOID_P EQUAL 8)
  target_sources(XRayTHMLister PRIVATE lister/exports64.def)
  set_target_properties(XRayTHMLister PROPERTIES SUFFIX ".wlx64")
 else()
  target_sources(XRayTHMLister PRIVATE lister/exports32.def)
  set_target_properties(XRayTHMLister PROPERTIES SUFFIX ".wlx")
 endif()
endif()

add_executable(thm_info thm_info.cpp thm_document.cpp thm_decoder.cpp editor/texture_model.cpp)
add_executable(test_document test_document.cpp thm_document.cpp thm_decoder.cpp editor/texture_model.cpp)
add_test(NAME document COMMAND test_document)

target_include_directories(thm_info PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_include_directories(test_document PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
